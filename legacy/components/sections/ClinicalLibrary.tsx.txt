interface ClinicalLibraryProps {
  currentLang?: Lang;
  userTier?: UserTier;
}

export default function ClinicalLibrary({
  currentLang = 'fr',
  userTier = 'none',
}: ClinicalLibraryProps) {
  const t = I18N_DATA[currentLang];
  const isRtl = currentLang === 'ar';

  const [activeTab, setActiveTab] = useState<CatDoc>('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [downloadCounts, setDownloadCounts] = useState<Record<string, number>>(() =>
    Object.fromEntries(CLINICAL_DOCUMENTS.map((doc) => [doc.id, doc.downloads]))
  );
  const [gatedDocNotice, setGatedDocNotice] = useState<ClinicalDocument | null>(null);
  const [downloadSuccessDoc, setDownloadSuccessDoc] = useState<ClinicalDocument | null>(null);

  const filteredDocuments = useMemo(() => {
    return CLINICAL_DOCUMENTS.filter((doc) => {
      const matchesCategory = activeTab === 'all' || doc.category === activeTab;
      const title = doc.title[currentLang].toLowerCase();
      const author = doc.author[currentLang].toLowerCase();
      const query = searchQuery.trim().toLowerCase();
      const matchesSearch = !query || title.includes(query) || author.includes(query);
      return matchesCategory && matchesSearch;
    });
  }, [activeTab, searchQuery, currentLang]);

  const handleDownloadAttempt = (doc: ClinicalDocument) => {
    const isMemberAuthenticated = userTier !== 'none';

    if (doc.isGated && !isMemberAuthenticated) {
      setGatedDocNotice(doc);
      return;
    }

    setDownloadCounts((prev) => ({
      ...prev,
      [doc.id]: (prev[doc.id] || 0) + 1,
    }));

    setDownloadSuccessDoc(doc);
    setTimeout(() => {
      setDownloadSuccessDoc(null);
    }, 4500);
  };

  return (
    <section
      id="library"
      aria-label={t.navLibrary}
      className="relative z-10 bg-[#FAF7F2] py-16 sm:py-24 border-b border-[#E8DFD5] scroll-mt-20"
    >
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
        
        {/* Section Header */}
        <div className="flex flex-col md:flex-row md:items-end justify-between mb-12 gap-6">
          <div className="max-w-2xl">
            <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-[#7A9B76]/15 border border-[#7A9B76]/30 text-xs font-semibold text-[#5A8A82] mb-3">
              <BookOpen className="w-3.5 h-3.5 stroke-[2]" />
              <span>04 // PROTOCOLES & MATRICES CLINIQUES</span>
            </div>

            <h2 className="text-3xl sm:text-4xl font-extrabold text-[#2A2622] tracking-tight">
              {currentLang === 'ar'
                ? 'المكتبة السريرية وشبكات التقييم المقننة'
                : 'Bibliothèque Clinique & Outils de Bilan'}
            </h2>

            <p className="mt-3 text-sm sm:text-base text-[#2A2622]/70 leading-relaxed font-normal">
              {currentLang === 'ar'
                ? 'فضاء توثيقي معتمد يجمع بطاريات الاختبار الحركي، النماذج المقننة، والمناشير المنظمة للمهنة.'
                : 'Accès sécurisé aux épreuves d’évaluation psychomotrice étalonnées, protocoles neurodéveloppementaux et circulaires officielles.'}
            </p>
          </div>

          {/* Quick Search Field */}
          <div className="w-full md:w-80 relative">
            <Search className="w-4 h-4 absolute left-3.5 top-1/2 -translate-y-1/2 text-[#2A2622]/40" />
            <input
              type="text"
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              placeholder={currentLang === 'ar' ? 'بحث في الاختبارات والمراجع...' : 'Rechercher un test ou protocole...'}
              className="w-full bg-[#F4EFEA] border border-[#E8DFD5] rounded-2xl pl-10 pr-4 py-3 text-xs text-[#2A2622] placeholder-[#2A2622]/40 outline-none focus:border-[#5A8A82] transition-colors"
            />
          </div>
        </div>

        {/* Sensory Category Filter Tabs */}
        <div className="flex flex-wrap items-center gap-2 mb-8 border-b border-[#E8DFD5] pb-4">
          {[
            { id: 'all', fr: 'Tous les documents', ar: 'كافة المراجع', en: 'All Documents' },
            { id: 'bilan', fr: 'Épreuves & Tests', ar: 'اختبارات وتقييمات', en: 'Batteries & Tests' },
            { id: 'protocole', fr: 'Protocoles Cliniques', ar: 'بروتوكولات سريرية', en: 'Protocols' },
            { id: 'officiel', fr: 'Circulaires Officielles', ar: 'مناشير وزارية', en: 'Official Directives' },
          ].map((tab) => (
            <button
              key={tab.id}
              type="button"
              onClick={() => setActiveTab(tab.id as CatDoc)}
              className={`min-h-[44px] px-4 py-2 rounded-xl text-xs font-semibold transition-all duration-300 ${
                activeTab === tab.id
                  ? 'bg-[#5A8A82] text-[#FAF7F2] shadow-sensory-sm'
                  : 'bg-[#F4EFEA] text-[#2A2622]/70 hover:bg-[#E8DFD5]'
              }`}
            >
              {tab[currentLang]}
            </button>
          ))}
        </div>

        {/* Document Grid */}
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6 items-stretch">
          {filteredDocuments.map((doc) => (
            <div
              key={doc.id}
              className="flex flex-col justify-between rounded-3xl border border-[#E8DFD5] bg-[#F4EFEA] p-6 shadow-sensory-sm transition-all duration-300 hover:border-[#7A9B76]/50"
            >
              <div>
                {/* Format and Security Status Tag */}
                <div className="flex items-center justify-between gap-2 mb-4">
                  <span className="px-2.5 py-1 rounded-md text-[10px] font-mono font-bold bg-[#FAF7F2] text-[#5A8A82] border border-[#E8DFD5]">
                    {doc.format}
                  </span>

                  {doc.isGated ? (
                    <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-[10px] font-bold bg-[#C67B5C]/15 text-[#C67B5C] border border-[#C67B5C]/30">
                      <Lock className="w-2.5 h-2.5 stroke-[2.5]" />
                      <span>{currentLang === 'ar' ? 'مقيد للمنخرطين' : 'Membres CSNPT'}</span>
                    </span>
                  ) : (
                    <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-[10px] font-bold bg-[#7A9B76]/15 text-[#5A8A82] border border-[#7A9B76]/30">
                      <FileCheck2 className="w-2.5 h-2.5 stroke-[2.5]" />
                      <span>{currentLang === 'ar' ? 'ولوج حر' : 'Accès Public'}</span>
                    </span>
                  )}
                </div>

                {/* Title and Origin */}
                <h3 className="text-base font-bold text-[#2A2622] leading-snug mb-2">
                  {doc.title[currentLang]}
                </h3>

                <p className="text-xs text-[#2A2622]/65 mb-6 font-normal leading-relaxed">
                  {doc.author[currentLang]}
                </p>
              </div>

              {/* File Specs and Download Trigger */}
              <div className="pt-4 border-t border-[#E8DFD5]">
                <div className="flex justify-between items-center text-[11px] font-mono text-[#2A2622]/60 mb-3" dir="ltr">
                  <span>{doc.size}</span>
                  <span>{downloadCounts[doc.id]} téléchargements</span>
                </div>

                <button
                  type="button"
                  onClick={() => handleDownloadAttempt(doc)}
                  className="w-full min-h-[48px] rounded-xl bg-[#FAF7F2] hover:bg-[#5A8A82] text-[#2A2622] hover:text-[#FAF7F2] border border-[#E8DFD5] text-xs font-semibold transition-all duration-300 flex items-center justify-center gap-2 shadow-sensory-sm"
                >
                  <Download className="w-4 h-4 stroke-[2]" />
                  <span>{currentLang === 'ar' ? 'تحميل الوثيقة' : 'Télécharger'}</span>
                </button>
              </div>
            </div>
          ))}
        </div>

        {/* Inline Feedback: Download Success */}
        {downloadSuccessDoc && (
          <div className="fixed bottom-6 right-6 z-50 rounded-2xl bg-[#5A8A82] text-[#FAF7F2] p-4 shadow-sensory-md flex items-center gap-3 border border-[#FAF7F2]/20 animate-in">
            <CheckCircle2 className="w-5 h-5 shrink-0" />
            <div className="text-xs">
              <span className="font-bold block">Document délivré avec succès</span>
              <span className="opacity-90">{downloadSuccessDoc.fileName}</span>
            </div>
          </div>
        )}

      </div>

      {/* ── MODAL: STATUTORY RESTRICTION NOTICE ── */}
      {gatedDocNotice && (
        <div
          role="dialog"
          aria-modal="true"
          className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-[#2A2622]/40 backdrop-blur-sm animate-in"
        >
          <div className="w-full max-w-md rounded-3xl bg-[#FAF7F2] border border-[#E8DFD5] p-6 sm:p-8 shadow-sensory-lg relative text-[#2A2622]">
            <button
              type="button"
              onClick={() => setGatedDocNotice(null)}
              className="absolute top-5 right-5 p-2 rounded-xl text-[#2A2622]/50 hover:bg-[#F4EFEA] transition-colors"
              aria-label="Fermer"
            >
              <X className="w-5 h-5" />
            </button>

            <div className="w-12 h-12 rounded-2xl bg-[#C67B5C]/15 text-[#C67B5C] flex items-center justify-center mb-4 border border-[#C67B5C]/30">
              <Lock className="w-6 h-6 stroke-[2]" />
            </div>

            <span className="text-[11px] font-bold text-[#C67B5C] uppercase tracking-wider block mb-1">
              Protocole Réservé
            </span>

            <h3 className="text-xl font-bold tracking-tight mb-2">
              Épreuve Diagnostique Réglementée
            </h3>

            <p className="text-xs text-[#2A2622]/75 leading-relaxed mb-6 font-normal">
              Conformément à la déontologie syndicale de la CSNPT (UTICA), l’accès complet aux matrices de passation et barèmes cliniques officiels (notamment <strong>{gatedDocNotice.title.fr}</strong>) est strictement réservé aux praticiens inscrits au registre national à jour de cotisation.
            </p>

            <div className="space-y-2.5">
              <a
                href="#pricing"
                onClick={() => setGatedDocNotice(null)}
                className="w-full min-h-[48px] rounded-xl bg-[#C67B5C] hover:bg-[#B56D4E] text-[#FAF7F2] text-xs font-semibold flex items-center justify-center gap-2 transition-all shadow-sensory-sm"
              >
                <span>Régulariser mon adhésion syndicale</span>
                <ArrowRight className="w-4 h-4 stroke-[2]" />
              </a>

              <button
                type="button"
                onClick={() => setGatedDocNotice(null)}
                className="w-full min-h-[44px] rounded-xl text-xs font-semibold text-[#2A2622]/60 hover:bg-[#F4EFEA] transition-colors"
              >
                Consulter les ressources publiques
              </button>
            </div>
          </div>
        </div>
      )}
    </section>
  );
}
